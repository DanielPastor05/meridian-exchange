cmake_minimum_required(VERSION 3.20)
project(meridian_exchange VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MERIDIAN_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)

add_library(meridian src/engine.cpp src/journal.cpp src/service.cpp src/protocol.cpp)
target_include_directories(meridian PUBLIC include)
if(MSVC)
  target_compile_options(meridian PUBLIC /W4 /WX /permissive- /utf-8)
  if(MERIDIAN_SANITIZE)
    message(FATAL_ERROR "MERIDIAN_SANITIZE requires GCC or Clang with address/undefined sanitizers")
  endif()
else()
  target_compile_options(meridian PUBLIC -Wall -Wextra -Wpedantic -Werror)
  if(MERIDIAN_SANITIZE)
    target_compile_options(meridian PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(meridian PUBLIC -fsanitize=address,undefined)
  endif()
endif()

find_package(Threads REQUIRED)
add_executable(exchange_server src/server.cpp src/net.cpp)
target_link_libraries(exchange_server PRIVATE meridian Threads::Threads)
if(WIN32)
  target_link_libraries(exchange_server PRIVATE ws2_32)
endif()

add_executable(exchange src/main.cpp)
target_link_libraries(exchange PRIVATE meridian)
add_executable(exchange_tests tests/tests.cpp)
target_link_libraries(exchange_tests PRIVATE meridian)
add_executable(exchange_bench bench/bench.cpp)
target_link_libraries(exchange_bench PRIVATE meridian)

add_executable(exchange_service_tests tests/service_tests.cpp)
target_link_libraries(exchange_service_tests PRIVATE meridian)

find_program(NODE_EXECUTABLE node REQUIRED)
enable_testing()
add_test(NAME tcp_integration COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/network.mjs" $<TARGET_FILE:exchange_server>)
add_test(NAME crash_recovery COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/crash.mjs" $<TARGET_FILE:exchange_server>)
set_tests_properties(tcp_integration crash_recovery PROPERTIES TIMEOUT 120)
add_test(NAME accounts_and_sessions COMMAND exchange_service_tests)
add_test(NAME correctness COMMAND exchange_tests)
add_test(NAME benchmark_smoke COMMAND exchange_bench 1000 1)
add_test(NAME cli_roundtrip COMMAND ${CMAKE_COMMAND}
  -DEXCHANGE=$<TARGET_FILE:exchange> -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-test
  -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)

add_executable(exchange_protocol_tests tests/protocol_tests.cpp)
target_link_libraries(exchange_protocol_tests PRIVATE meridian)
add_test(NAME protocol_boundaries COMMAND exchange_protocol_tests)
option(MERIDIAN_FUZZ "Build Clang libFuzzer protocol target" OFF)
if(MERIDIAN_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MERIDIAN_SANITIZE)
    message(FATAL_ERROR "Fuzzing requires Clang and MERIDIAN_SANITIZE=ON")
  endif()
  target_compile_options(meridian PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(exchange_fuzz tests/fuzz_protocol.cpp)
  target_link_libraries(exchange_fuzz PRIVATE meridian)
  target_compile_options(exchange_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(exchange_fuzz PRIVATE -fsanitize=fuzzer)
endif()

if(NOT MERIDIAN_SANITIZE)
  add_executable(exchange_profile bench/profile.cpp)
  target_link_libraries(exchange_profile PRIVATE meridian)
endif()
