cmake_minimum_required(VERSION 3.20)
project(meridian_exchange VERSION 1.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(MERIDIAN_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(MERIDIAN_TSAN "Enable thread sanitizer (separate from ASan)" OFF)
option(MERIDIAN_FUZZ "Build Clang libFuzzer protocol target" OFF)
option(MERIDIAN_BUILD_BENCHMARKS "Build benchmark tools" ON)
if(MERIDIAN_TSAN AND MERIDIAN_SANITIZE)
  message(FATAL_ERROR "ThreadSanitizer and AddressSanitizer require separate builds")
endif()
if(MSVC AND (MERIDIAN_SANITIZE OR MERIDIAN_TSAN))
  message(FATAL_ERROR "Sanitizers require GCC or Clang on a supported platform")
endif()

function(meridian_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()

add_library(meridian src/engine.cpp src/journal.cpp src/service.cpp src/protocol.cpp src/snapshot.cpp)
target_include_directories(meridian PUBLIC include)
meridian_warnings(meridian)
if(MERIDIAN_SANITIZE)
  target_compile_options(meridian PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(meridian PUBLIC -fsanitize=address,undefined)
elseif(MERIDIAN_TSAN)
  target_compile_options(meridian PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
  target_link_options(meridian PUBLIC -fsanitize=thread)
endif()

find_package(Threads REQUIRED)
add_library(meridian_net src/net.cpp)
target_link_libraries(meridian_net PUBLIC meridian Threads::Threads)
meridian_warnings(meridian_net)
if(WIN32)
  target_link_libraries(meridian_net PUBLIC ws2_32)
endif()

add_executable(exchange_server src/server.cpp)
target_link_libraries(exchange_server PRIVATE meridian_net)
meridian_warnings(exchange_server)
add_executable(exchange src/main.cpp)
target_link_libraries(exchange PRIVATE meridian)
meridian_warnings(exchange)
configure_file(include/meridian/version.hpp.in generated/meridian/version.hpp @ONLY)
target_include_directories(exchange PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated")

if(MERIDIAN_BUILD_BENCHMARKS)
  add_executable(exchange_bench bench/bench.cpp)
  target_link_libraries(exchange_bench PRIVATE meridian)
  meridian_warnings(exchange_bench)
  if(NOT MERIDIAN_SANITIZE AND NOT MERIDIAN_TSAN)
    add_executable(exchange_profile bench/profile.cpp)
    target_link_libraries(exchange_profile PRIVATE meridian)
    meridian_warnings(exchange_profile)
  endif()
endif()

if(BUILD_TESTING)
  find_program(NODE_EXECUTABLE node REQUIRED)
  foreach(suite IN ITEMS tests service_tests protocol_tests net_tests snapshot_tests storage_tests)
    add_executable(exchange_${suite} tests/${suite}.cpp)
    target_link_libraries(exchange_${suite} PRIVATE meridian_net)
    meridian_warnings(exchange_${suite})
  endforeach()
  add_test(NAME correctness COMMAND exchange_tests)
  add_test(NAME accounts_and_sessions COMMAND exchange_service_tests)
  add_test(NAME protocol_boundaries COMMAND exchange_protocol_tests)
  add_test(NAME accept_errors COMMAND exchange_net_tests)
  add_test(NAME book_snapshots COMMAND exchange_snapshot_tests)
  add_test(NAME storage_failures COMMAND exchange_storage_tests)
  add_test(NAME process_helpers COMMAND "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/tests/process.mjs")
  add_test(NAME client_validation COMMAND "${NODE_EXECUTABLE}" --test "${CMAKE_CURRENT_SOURCE_DIR}/tests/client.mjs")
  add_test(NAME tcp_integration COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/network.mjs" $<TARGET_FILE:exchange_server>)
  add_test(NAME crash_recovery COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/crash.mjs" $<TARGET_FILE:exchange_server>)
  add_test(NAME session_concurrency COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/concurrency.mjs" $<TARGET_FILE:exchange_server>)
  set_tests_properties(tcp_integration crash_recovery session_concurrency PROPERTIES TIMEOUT 120)
  add_test(NAME cli_roundtrip COMMAND ${CMAKE_COMMAND}
    -DEXCHANGE=$<TARGET_FILE:exchange> -DTEST_DIR=${CMAKE_CURRENT_BINARY_DIR}/cli-test
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
  if(MERIDIAN_BUILD_BENCHMARKS)
    add_test(NAME benchmark_smoke COMMAND exchange_bench 1000 1)
  endif()
endif()

if(MERIDIAN_FUZZ)
  if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang" OR NOT MERIDIAN_SANITIZE)
    message(FATAL_ERROR "Fuzzing requires Clang and MERIDIAN_SANITIZE=ON")
  endif()
  target_compile_options(meridian PRIVATE -fsanitize=fuzzer-no-link)
  add_executable(exchange_fuzz tests/fuzz_protocol.cpp)
  target_link_libraries(exchange_fuzz PRIVATE meridian)
  target_compile_options(exchange_fuzz PRIVATE -fsanitize=fuzzer)
  target_link_options(exchange_fuzz PRIVATE -fsanitize=fuzzer)
endif()
